#include <tunables/global>

profile replicarr flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Python runtime
  /usr/bin/python3* mr,
  /usr/lib/python3*/ r,
  /usr/lib/python3*/** mr,
  /usr/local/lib/python3*/ r,
  /usr/local/lib/python3*/** mr,

  # App files
  /app/ r,
  /app/** mr,

  # Persistent data volume
  /data/ r,
  /data/** rw,

  # Capabilities required by s6-overlay-suexec (privilege drop)
  capability setuid,
  capability setgid,
  capability dac_override,
  capability dac_read_search,
  capability chown,
  capability fowner,

  # s6-overlay v3 init and commands
  /init rix,
  /command/ r,
  /command/** rix,
  /package/ r,
  /package/** rix,
  /etc/s6-overlay/ r,
  /etc/s6-overlay/** r,

  # /run — parent dir needs rw so s6 can mkdir /run/s6 and /run/service.
  # CONFIRMED BROKEN as rw-only: on container restart, s6-overlay's stage0
  # re-execs into /run/s6/basedir/bin/init (a copy of s6-overlay's own init
  # it stages there at boot). rw does not grant execute, so that re-exec was
  # denied and the container looped forever on "Permission denied" until
  # apparmor was disabled again. If re-attempting this profile, this line
  # needs to become `/run/** rwix,` (or a scoped rix rule specifically for
  # /run/s6/basedir/**) — and it must be verified through an actual add-on
  # *restart*, not just a first boot, since that's the path that broke.
  /run/ rw,
  /run/** rw,
  /var/run/ rw,
  /var/run/** rw,

  # proc — s6 reads /proc/self during suexec
  @{PROC}/self/** r,
  @{PROC}/@{pid}/** r,

  # bashio / contenv
  /usr/bin/with-contenv rix,
  /usr/bin/bashio rix,
  /usr/bin/s6* rix,
  /usr/lib/bashio/ r,
  /usr/lib/bashio/** r,

  # Shell and the specific utilities bashio/run scripts invoke.
  # Alpine's coreutils are a busybox multi-call binary — most of `cat`,
  # `mkdir`, `chmod`, `grep`, `sed`, `cut`, `sleep`, `date`, etc. resolve to
  # /bin/busybox, not separate files, so one rule covers them.
  /bin/busybox rix,
  /bin/bash rix,
  /bin/sh rix,
  /usr/bin/env rix,
  /usr/bin/jq rix,
  /usr/bin/curl rix,
  /usr/bin/tempio rix,

  # Temp files (bashio writes config here)
  /tmp/ rw,
  /tmp/** rw,

  # Network (outbound HTTP to Syncthing instances)
  network tcp,
  network udp,

  # Deny everything else
  deny /proc/sysrq-trigger rw,
  deny /sys/firmware/** rw,
}
